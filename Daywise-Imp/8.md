# 1. Install Docker, Kind, and Kubectl
sudo apt-get update && sudo apt-get install -y docker.io curl && sudo usermod -aG docker $USER
curl -Lo ./kind https://kind.sigs.k8s.io/dl/v0.32.0/kind-linux-amd64 && chmod +x ./kind && sudo mv ./kind /usr/local/bin/kind
curl -LO "https://dl.k8s.io/release/v1.31.4/bin/linux/amd64/kubectl" && chmod +x kubectl && sudo mv kubectl /usr/local/bin/

# 2. Create Clusters
kind create cluster --name my-first-cluster --config kind-cluster.yaml
kind create cluster --name my-second-cluster --image kindest/node:v1.31.4

# 3. Context Switching & Verification
kubectl config get-contexts
kubectl config use-context kind-my-second-cluster
kubectl get nodes
